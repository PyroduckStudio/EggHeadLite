// Imported as a shader only after HDRP is installed, keeping Built-in/URP imports clean.
Shader "PyroDuck/EggHeads/HDRP Sprite Unlit"
{
    Properties
    {
        [PerRendererData] _MainTex ("Sprite Texture", 2D) = "white" {}
        _Color ("Tint", Color) = (1,1,1,1)
        [HideInInspector] _EggHeadsRendererColor ("Renderer Color", Color) = (1,1,1,1)
        [HideInInspector] _EggHeadsIsSprite ("Sprite Renderer", Float) = 0
        [PerRendererData] _AlphaTex ("External Alpha", 2D) = "white" {}
        [PerRendererData] _EnableExternalAlpha ("External Alpha Enabled", Float) = 0
    }
    SubShader
    {
        Tags { "RenderPipeline"="HDRenderPipeline" "Queue"="Transparent" "RenderType"="Transparent" "CanUseSpriteAtlas"="True" }
        Pass
        {
            Name "ForwardOnly"
            Tags { "LightMode"="ForwardOnly" }
            Cull Off
            ZWrite Off
            Blend SrcAlpha OneMinusSrcAlpha, One OneMinusSrcAlpha
            HLSLPROGRAM
            #pragma target 4.5
            #pragma vertex Vert
            #pragma fragment Frag
            #include "Packages/com.unity.render-pipelines.core/ShaderLibrary/Common.hlsl"
            #include "Packages/com.unity.render-pipelines.high-definition/Runtime/ShaderLibrary/ShaderVariables.hlsl"
            TEXTURE2D(_MainTex); SAMPLER(sampler_MainTex);
            TEXTURE2D(_AlphaTex); SAMPLER(sampler_AlphaTex);
            CBUFFER_START(UnityPerMaterial)
                float4 _Color;
                float4 _MainTex_ST;
                float _EnableExternalAlpha;
            CBUFFER_END
            // Per-renderer uniforms deliberately exclude this sprite shader from
            // the SRP Batcher, which cannot supply HDRP SpriteRenderer overrides.
            float4 _EggHeadsRendererColor;
            float _EggHeadsIsSprite;
            struct Attributes { float3 positionOS : POSITION; float2 uv : TEXCOORD0; float4 color : COLOR; };
            struct Varyings { float4 positionCS : SV_POSITION; float2 uv : TEXCOORD0; float4 color : COLOR; };
            Varyings Vert(Attributes input)
            {
                Varyings output;
                // HDRP's generic SpriteRenderer draw already flips the sprite geometry.
                output.positionCS = TransformObjectToHClip(input.positionOS);
                output.uv = input.uv * _MainTex_ST.xy + _MainTex_ST.zw;
                output.color = _Color * lerp(input.color, _EggHeadsRendererColor, _EggHeadsIsSprite);
                return output;
            }
            float4 Frag(Varyings input) : SV_Target
            {
                float4 color = SAMPLE_TEXTURE2D(_MainTex, sampler_MainTex, input.uv);
                color.a = lerp(color.a, SAMPLE_TEXTURE2D(_AlphaTex, sampler_AlphaTex, input.uv).r, _EnableExternalAlpha);
                color *= input.color;
                // HDRP resolves pre-exposed color after the forward pass.
                color.rgb *= GetCurrentExposureMultiplier();
                return color;
            }
            ENDHLSL
        }
    }
    Fallback Off
}
